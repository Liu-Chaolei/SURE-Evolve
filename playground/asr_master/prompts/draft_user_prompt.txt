## Example wrapper script structure

```python
import sys
import os
import subprocess
import re
import random

# CRITICAL: Use icefall conda environment Python
icefall_python = "/home/liuchaolei/anaconda3/envs/icefall/bin/python"

# Use RELATIVE paths directly (NO os.path.dirname(__file__) needed!)
icefall_root = "./icefall_root"
recipe_dir = "./icefall_recipe"
data_dir = "./icefall_data"

# Set up paths
sys.path.insert(0, icefall_root)
sys.path.insert(0, recipe_dir)
os.chdir(recipe_dir)

# The cuts manifests store feature paths relative to the recipe as data/fbank/...
if not os.path.exists("data"):
    os.symlink("../icefall_data", "data")

# Experiment directory
exp_dir = "./exp_asr_opt"

# Manifest and BPE model paths
# CRITICAL: train.py expects librispeech_cuts_*.jsonl.gz under --manifest-dir.
# Use ./data because embedded Lhotse feature paths also start with data/fbank.
manifest_dir = "./data/fbank"
bpe_model = "./data/lang_bpe_500/bpe.model"

def detect_gpu_mem_gb():
    try:
        probe = subprocess.run(
            [
                icefall_python,
                "-c",
                "import torch; print(torch.cuda.get_device_properties(0).total_memory / (1024 ** 3) if torch.cuda.is_available() else 0)",
            ],
            capture_output=True,
            text=True,
            timeout=20,
        )
        if probe.returncode == 0:
            return float(probe.stdout.strip() or "0")
    except Exception:
        pass
    return 0.0

def choose_max_duration():
    override = os.environ.get("ASR_MAX_DURATION")
    if override:
        return int(float(override))
    mem_gb = detect_gpu_mem_gb()
    if mem_gb >= 80:
        return 2200
    if mem_gb >= 48:
        return 1700
    if mem_gb >= 32:
        return 1400
    if mem_gb >= 24:
        return 1000
    if mem_gb > 0:
        return 800
    return 1000

def infer_world_size():
    override = os.environ.get("ASR_WORLD_SIZE")
    if override:
        return max(1, int(override))
    visible = os.environ.get("CUDA_VISIBLE_DEVICES", "").strip()
    if visible and visible not in {"-1", "none", "None"}:
        return max(1, len([x for x in visible.split(",") if x.strip()]))
    return 1

def choose_master_port():
    override = os.environ.get("ASR_MASTER_PORT")
    if override:
        return override
    return str(12354 + random.randint(0, 2000))

def next_oom_max_duration(value):
    if value > 1700:
        return 1700
    if value > 1400:
        return 1400
    if value > 1000:
        return 1000
    if value > 800:
        return 800
    if value > 500:
        return 500
    if value > 300:
        return 300
    if value > 200:
        return 200
    return None

max_duration = choose_max_duration()
world_size = infer_world_size()
master_port = choose_master_port()

# Step 1: Train. Use fp16 first for speed, then fall back to fp32 on non-finite/grad-scale instability.
base_lr = "0.02"
use_fp16 = "1"
while True:
    train_cmd = [
        icefall_python, "train.py",
        "--world-size", str(world_size),
        "--master-port", master_port,
        "--num-epochs", "1",
        "--start-epoch", "1",
        "--use-fp16", use_fp16,
        "--exp-dir", exp_dir,
        "--full-libri", "1",
        "--max-duration", str(max_duration),
        "--base-lr", base_lr,
        "--manifest-dir", manifest_dir,
        "--bpe-model", bpe_model,
    ]
    train_result = subprocess.run(train_cmd, capture_output=True, text=True)
    if train_result.returncode == 0:
        break
    train_output = train_result.stdout + train_result.stderr
    lower_output = train_output.lower()
    lowered = next_oom_max_duration(max_duration)
    if ("cuda out of memory" in lower_output or "out of memory" in lower_output) and lowered is not None:
        max_duration = lowered
        continue
    if use_fp16 == "1" and (
        "grad_scale is too small" in lower_output
        or "loss=inf" in lower_output
        or "metric=nan" in lower_output
        or "not finite" in lower_output
    ):
        use_fp16 = "0"
        continue
    print(train_output)
    train_result.check_returncode()

# Step 2: Decode
decode_cmd = [
    icefall_python, "decode.py",
    "--epoch", "1",
    "--avg", "1",
    "--use-averaged-model", "0",
    "--exp-dir", exp_dir,
    "--max-duration", str(max(600, min(max_duration, 1200))),
    "--decoding-method", "modified_beam_search",
]
result = subprocess.run(decode_cmd, capture_output=True, text=True)

# Step 3: Parse WER from decode output
output = result.stdout + result.stderr
wer_match = re.findall(r'(?:WER|wer)[\s:=]+(\d+\.?\d*)', output)
if wer_match:
    final_wer = float(wer_match[-1])
    print(f"\\boxed{{{{{{final_wer}}}}}}")
else:
    print("WER not found in decode output")
```
